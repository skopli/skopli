Install the SDK for your language, then make the four calls: detect the harnesses on this machine, read their session logs, roll the events up, and price the result. Step 2 is a quickstart that prints each priced row.

Only the Price call touches the network, and [Pricing basics](/skopli/guide/pricing-basics) covers its sources, cache, offline use, and misses.

## Step 1: install

No package is published yet. The registry commands below work after the [first release](https://github.com/skopli/skopli/releases), and until then each SDK builds from the repository. The Java and Swift tabs already include it from a checkout beside your project. The TypeScript package needs Node 24 or newer and ships ESM only. Java, Swift, and C link the C library, built from the repository root with `cargo build -p skopli-capi --release`.

## Step 2: quickstart

<Tabs syncKey="lang">
  <Tab label="TypeScript">
    ```ts
    import { detectHarnesses, readUsage, rollup, createPricing } from "skopli";

    // what harnesses have local data?
    const { supported, unsupported } = detectHarnesses();

    // read usage events (all detected harnesses, or pick explicitly)
    const { events, diagnostics } = await readUsage({
      harnesses: ["claude", "opencode", "codex"],
      since: "2026-08-01",
    });

    // aggregate
    const byModel = rollup(events, { by: "model" });
    const byDay = rollup(events, { by: "day", tz: "Europe/London" });

    // price against market catalogs
    const pricing = createPricing();
    const priced = await pricing.priceRollups(byModel);
    for (const row of priced) {
      if (row.pricing.priced) {
        console.log(row.key, row.pricing.usd.toFixed(2),
            `(${row.pricing.source})`);
      } else {
        console.log(row.key, "unpriced; tried",
            row.pricing.attempted.join(", "));
      }
    }
    ```

  </Tab>
  <Tab label="Python">
    ```python
    import skopli

    # what harnesses have local data?
    detection = skopli.detect_harnesses()

    # read usage events (all detected harnesses, or pick explicitly)
    result = skopli.read_usage(
        harnesses=["claude", "opencode", "codex"],
        since="2026-08-01",
    )

    # aggregate
    by_model = skopli.rollup(result.events, "model")
    by_day = skopli.rollup(result.events, "day", tz="Europe/London")

    # price against market catalogs
    pricing = skopli.create_pricing()
    for row in pricing.price_rollups(by_model):
        if row.pricing.priced:
            print(row.rollup.key, round(row.usd, 2),
                f"({row.pricing.source})")
        else:
            print(row.rollup.key, "unpriced; tried",
                ", ".join(row.pricing.attempted))
    ```

  </Tab>
  <Tab label="Rust">
    `skopli-core` exposes the aggregation and pricing primitives directly. Roll up a batch of `UsageEvent`s and price a token bundle:

    ```rust
    use skopli_core::rollup::{rollup, RollupBy, RollupOptions};
    use skopli_core::pricing::{cost_usd, ModelPrice};
    use skopli_core::types::TokenCounts;

    // aggregate events by model (timezone-aware day bucketing for
    // `RollupBy::Day`)
    let by_model = rollup(&events, &RollupOptions::new(RollupBy::Model));
    for bucket in &by_model {
        println!("{}: {} events", bucket.key, bucket.events);
    }

    // price a token bundle at a flat rate (USD per million tokens)
    let tokens = TokenCounts {
        input: 1_000,
        output: 500,
        ..Default::default()
    };
    let usd = cost_usd(&tokens, &ModelPrice::flat(1.25, 10.0));
    println!("{usd:.5}");
    ```

  <Tab label="Java">
    ```java
    import com.skopli.*;
    import java.util.List;

    // what harnesses have local data?
    Detection detection = Skopli.detectHarnesses(DetectOptions.defaults());

    // read usage events (all detected harnesses, or pick explicitly)
    ReadUsageResult result = Skopli.readUsage(ReadUsageOptions.builder()
        .harness(Harness.CLAUDE)
        .harness(Harness.OPENCODE)
        .since("2026-08-01")
        .build());

    // aggregate
    List<Rollup> byModel = Skopli.rollup(result.events(),
        RollupOptions.by(RollupBy.MODEL));

    // price against market catalogs
    PricingOptions defaults = PricingOptions.builder().build();
    try (Pricing pricing = Skopli.createPricing(defaults)) {
        for (PricedRollup row : pricing.priceRollups(byModel)) {
            System.out.println(row.rollup().key() + " " + row.pricing());
        }
    }
    ```

    ```csharp
    using Skopli;

    // what harnesses have local data?
    Detection detection = SkopliClient.DetectHarnesses();

    // read usage events (all detected harnesses, or pick explicitly)
    ReadUsageResult usage = SkopliClient.ReadUsage(new ReadUsageOptions {

    // aggregate
    IReadOnlyList<Rollup> byModel =

            By = RollupBy.Model,
        });

    // price against market catalogs
    using Pricing pricing = SkopliClient.CreatePricing(new PricingOptions());
    IReadOnlyList<PricedRollup> priced = pricing.PriceRollups(byModel);
    ```

  <Tab label="Go">
    ```go
    import "github.com/skopli/skopli/sdks/go/skopli"

    // what harnesses have local data?
    detection, _ := skopli.DetectHarnesses(nil)

    // read usage events (all detected harnesses, or pick explicitly)
    result, _ := skopli.ReadUsage(skopli.ReadUsageOptions{
        Harnesses: []skopli.Harness{
            skopli.HarnessClaude,
            skopli.HarnessOpencode,
        },
        Since:     "2026-08-01",
    })

    // aggregate
    byModel, _ := skopli.Rollup(result.Events, skopli.RollupOptions{
        By: skopli.ByModel,
    })

    // price against market catalogs
    pricing, _ := skopli.NewPricing(skopli.PricingOptions{})
    defer pricing.Close()
    priced, _ := pricing.PriceRollups(byModel)
    for _, row := range priced {
        fmt.Println(row.Key, row.Pricing.Priced)
    }
    ```

  <Tab label="Swift">
    ```swift
    import Skopli

    // what harnesses have local data?
    let detection = try Skopli.detectHarnesses()

    // read usage events (all detected harnesses, or pick explicitly)
    let result = try Skopli.readUsage(
        options: ReadUsageOptions(harnesses: [.claude, .opencode],
            since: "2026-08-01"))

    // aggregate
    let byModel = try Skopli.rollup(events: result.events, by: .model)

    // price against market catalogs
    let pricing = try Skopli.createPricing()
    defer { pricing.close() }
    let priced = try pricing.priceRollups(rollups: byModel)
    for row in priced {
        print(row.key, row.pricing.priced)
    }
    ```

  <Tab label="C">
    The C ABI exchanges JSON buffers. Roll up an event array (`events_json`) and free the returned buffer:

    ```c
    #include <stdio.h>
    #include <string.h>
    #include "skopli.h"

    // events_json / events_len are a serialized UsageEvent[] from
    // ag_read_usage.
    AgBuf out = {0};
    const char *group_opts = "{\"by\":\"model\"}";
    AgStatus st = ag_rollup(events_json, events_len, group_opts,
        strlen(group_opts), &out);
    if (st == AG_STATUS_OK) {
        fwrite(out.ptr, 1, out.len, stdout); // Rollup[] as JSON
        ag_buf_free(out);
    }
    ```
